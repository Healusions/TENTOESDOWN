
add_namespace = IND_opening

##################################
# HIDDEN CAMPAIGN INITIALIZATION
##################################

country_event = {
	id = IND_opening.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = { has_country_flag = IND_campaign_initialized }
			}

			set_country_flag = IND_campaign_initialized

			# Core campaign variables. Keep all starting values in this one block.
			set_variable = { var = IND_federal_pressure value = 20 }
			set_variable = { var = IND_machine_control value = 50 }
			set_variable = { var = IND_guard_cohesion value = 50 }
			set_variable = { var = IND_consortium_exposure value = 5 }
			set_variable = { var = IND_underground value = 10 }

			# Temporary development safeguard. A later federal crisis can shorten this truce.
			set_truce = {
				target = USA
				days = 730
			}
			USA = {
				set_truce = {
					target = ROOT
					days = 730
				}
			}
		}
	}
}
